# StewardedNode Topology Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Add StewardedNode as a first-class entity flowing from Holochain DNA through elohim-storage to the shefa dashboard, with two seeded nodes and availability aggregation.

**Architecture:** Full protocol path — DNA entry → elohim-storage projection (SQLite + HTTP) → ts-rs generated types → Angular shefa dashboard. Stewardship is a many-to-many relationship (multiple humans steward each node with varying affinity). Natural language context is EPR-addressed content, not a text column. No doorway involvement — storage serves its own HTTP API.

**Tech Stack:** Rust (Holochain HDK, Diesel, hyper), TypeScript (Angular 19, ts-rs generated types), SQLite

**Design doc:** `genesis/plans/2026-03-12-stewarded-node-topology-design.md`

```sql
CREATE TABLE stewarded_nodes (
    id TEXT PRIMARY KEY NOT NULL,
    display_name TEXT NOT NULL,
    claim_status TEXT NOT NULL DEFAULT 'unclaimed',
    cpu_cores INTEGER NOT NULL DEFAULT 0,
    memory_gb INTEGER NOT NULL DEFAULT 0,
    storage_tb REAL NOT NULL DEFAULT 0.0,
    bandwidth_mbps INTEGER NOT NULL DEFAULT 0,
    steward_tier TEXT NOT NULL DEFAULT 'caretaker',
    custodian_opt_in INTEGER NOT NULL DEFAULT 1,
    region TEXT,
    context_epr_id TEXT,
    dht_anchor_hash TEXT,
    app_id TEXT NOT NULL DEFAULT 'shefa',
    created_at TEXT NOT NULL DEFAULT (datetime('now')),
    updated_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE INDEX idx_stewarded_nodes_claim_status ON stewarded_nodes(claim_status);
CREATE INDEX idx_stewarded_nodes_app_id ON stewarded_nodes(app_id);

-- NodeStewardship: many-to-many relationship between humans and nodes
-- A node has multiple stewards with varying affinity (like content stewardship)
CREATE TABLE node_stewardship (
    node_id TEXT NOT NULL REFERENCES stewarded_nodes(id),
    human_id TEXT NOT NULL REFERENCES humans(id),
    affinity_score REAL NOT NULL DEFAULT 0.0,
    relationship TEXT NOT NULL DEFAULT 'primary',
    context_epr_id TEXT,
    granted_at TEXT NOT NULL DEFAULT (datetime('now')),
    PRIMARY KEY (node_id, human_id)
);

CREATE INDEX idx_node_stewardship_human ON node_stewardship(human_id);
```

```rust
//! StewardedNode CRUD operations using Diesel
//!
//! Manages node registration and stewardship relationships for the shefa pillar.
//! Cryptographic node identity lives in the Holochain node-registry DNA;
//! this layer owns the fast, queryable storage projection.

use diesel::prelude::*;
use uuid::Uuid;

use super::diesel_schema::{node_stewardship, stewarded_nodes};
use super::models::{
    current_timestamp, NewNodeStewardship, NewStewardedNode, NodeStewardship, StewardedNode,
};
use crate::error::StorageError;

// ============================================================================
// Input Types
// ============================================================================

/// Input for registering a stewarded node
#[derive(Debug, Clone)]
pub struct CreateStewardedNodeInput {
    pub id: String,
    pub display_name: String,
    pub claim_status: String,
    pub cpu_cores: i32,
    pub memory_gb: i32,
    pub storage_tb: f64,
    pub bandwidth_mbps: i32,
    pub steward_tier: String,
    pub custodian_opt_in: bool,
    pub region: Option<String>,
    pub context_epr_id: Option<String>,
    pub app_id: String,
}

/// Input for creating a stewardship relationship
#[derive(Debug, Clone)]
pub struct CreateNodeStewardshipInput {
    pub node_id: String,
    pub human_id: String,
    pub affinity_score: f64,
    pub relationship: String,
    pub context_epr_id: Option<String>,
}

// ============================================================================
// CRUD Operations — StewardedNode
// ============================================================================

/// Register a new stewarded node.
pub fn create_stewarded_node(
    conn: &mut SqliteConnection,
    input: CreateStewardedNodeInput,
) -> Result<StewardedNode, StorageError> {
    let id = if input.id.is_empty() {
        Uuid::new_v4().to_string()
    } else {
        input.id
    };

    let new_node = NewStewardedNode {
        id: id.clone(),
        display_name: input.display_name,
        claim_status: input.claim_status,
        cpu_cores: input.cpu_cores,
        memory_gb: input.memory_gb,
        storage_tb: input.storage_tb,
        bandwidth_mbps: input.bandwidth_mbps,
        steward_tier: input.steward_tier,
        custodian_opt_in: if input.custodian_opt_in { 1 } else { 0 },
        region: input.region,
        context_epr_id: input.context_epr_id,
        dht_anchor_hash: None,
        app_id: input.app_id,
    };

    diesel::insert_into(stewarded_nodes::table)
        .values(&new_node)
        .execute(conn)
        .map_err(|e| StorageError::Internal(format!("Failed to insert stewarded node: {}", e)))?;

    get_stewarded_node_by_id(conn, &id)?
        .ok_or_else(|| StorageError::Internal("Node not found after insert".to_string()))
}

/// Get a stewarded node by ID.
pub fn get_stewarded_node_by_id(
    conn: &mut SqliteConnection,
    id: &str,
) -> Result<Option<StewardedNode>, StorageError> {
    stewarded_nodes::table
        .filter(stewarded_nodes::id.eq(id))
        .first::<StewardedNode>(conn)
        .optional()
        .map_err(|e| StorageError::Internal(format!("Failed to query node: {}", e)))
}

/// List all stewarded nodes, optionally filtered by claim status.
pub fn list_stewarded_nodes(
    conn: &mut SqliteConnection,
    claim_status: Option<&str>,
) -> Result<Vec<StewardedNode>, StorageError> {
    let mut query = stewarded_nodes::table.into_boxed();

    if let Some(status) = claim_status {
        query = query.filter(stewarded_nodes::claim_status.eq(status));
    }

    query
        .order(stewarded_nodes::created_at.desc())
        .load::<StewardedNode>(conn)
        .map_err(|e| StorageError::Internal(format!("Failed to list nodes: {}", e)))
}

// ============================================================================
// CRUD Operations — NodeStewardship
// ============================================================================

/// Create a stewardship relationship between a human and a node.
pub fn create_node_stewardship(
    conn: &mut SqliteConnection,
    input: CreateNodeStewardshipInput,
) -> Result<NodeStewardship, StorageError> {
    let new_stewardship = NewNodeStewardship {
        node_id: input.node_id.clone(),
        human_id: input.human_id.clone(),
        affinity_score: input.affinity_score,
        relationship: input.relationship,
        context_epr_id: input.context_epr_id,
    };

    diesel::insert_into(node_stewardship::table)
        .values(&new_stewardship)
        .execute(conn)
        .map_err(|e| {
            StorageError::Internal(format!("Failed to insert node stewardship: {}", e))
        })?;

    get_node_stewardship(conn, &input.node_id, &input.human_id)?
        .ok_or_else(|| StorageError::Internal("Stewardship not found after insert".to_string()))
}

/// Get a specific stewardship relationship.
pub fn get_node_stewardship(
    conn: &mut SqliteConnection,
    node_id: &str,
    human_id: &str,
) -> Result<Option<NodeStewardship>, StorageError> {
    node_stewardship::table
        .filter(node_stewardship::node_id.eq(node_id))
        .filter(node_stewardship::human_id.eq(human_id))
        .first::<NodeStewardship>(conn)
        .optional()
        .map_err(|e| StorageError::Internal(format!("Failed to query stewardship: {}", e)))
}

/// List all stewards for a given node.
pub fn list_stewards_for_node(
    conn: &mut SqliteConnection,
    node_id: &str,
) -> Result<Vec<NodeStewardship>, StorageError> {
    node_stewardship::table
        .filter(node_stewardship::node_id.eq(node_id))
        .order(node_stewardship::affinity_score.desc())
        .load::<NodeStewardship>(conn)
        .map_err(|e| StorageError::Internal(format!("Failed to list stewards: {}", e)))
}

/// List all nodes a human stewards.
pub fn list_nodes_for_human(
    conn: &mut SqliteConnection,
    human_id: &str,
) -> Result<Vec<NodeStewardship>, StorageError> {
    node_stewardship::table
        .filter(node_stewardship::human_id.eq(human_id))
        .order(node_stewardship::affinity_score.desc())
        .load::<NodeStewardship>(conn)
        .map_err(|e| StorageError::Internal(format!("Failed to list nodes for human: {}", e)))
}
```

```rust
/// POST /db/nodes — Register a stewarded node
async fn handle_create_node(
    req: Request<Incoming>,
    pool: &DbPool,
) -> Result<Response<Full<Bytes>>, StorageError> {
    let body = read_body(req).await?;
    let input: CreateStewardedNodeInputView = serde_json::from_slice(&body)
        .map_err(|e| StorageError::InvalidInput(format!("Invalid node input: {}", e)))?;

    let db_input: stewarded_nodes::CreateStewardedNodeInput = input.into();
    let mut conn = pool.get().map_err(|e| StorageError::Internal(e.to_string()))?;
    let node = stewarded_nodes::create_stewarded_node(&mut conn, db_input)?;

    let view: StewardedNodeView = node.into();
    response::created(&view)
}

/// GET /db/nodes — List stewarded nodes
async fn handle_list_nodes(
    req: &Request<Incoming>,
    pool: &DbPool,
) -> Result<Response<Full<Bytes>>, StorageError> {
    let query = parse_query_string(req.uri().query().unwrap_or(""));
    let claim_status = query.get("claimStatus").map(|s| s.as_str());

    let mut conn = pool.get().map_err(|e| StorageError::Internal(e.to_string()))?;
    let nodes = stewarded_nodes::list_stewarded_nodes(&mut conn, claim_status)?;

    let views: Vec<StewardedNodeView> = nodes.into_iter().map(|n| n.into()).collect();
    response::ok(&views)
}

/// GET /db/nodes/{id} — Get node with stewardship relationships
async fn handle_get_node(
    pool: &DbPool,
    node_id: &str,
) -> Result<Response<Full<Bytes>>, StorageError> {
    let mut conn = pool.get().map_err(|e| StorageError::Internal(e.to_string()))?;

    let node = stewarded_nodes::get_stewarded_node_by_id(&mut conn, node_id)?
        .ok_or_else(|| StorageError::NotFound(format!("Node not found: {}", node_id)))?;

    let stewards = stewarded_nodes::list_stewards_for_node(&mut conn, node_id)?;

    let mut view: StewardedNodeView = node.into();
    view.stewards = stewards.into_iter().map(|s| s.into()).collect();

    response::ok(&view)
}

/// POST /db/nodes/{id}/stewardship — Add stewardship relationship
async fn handle_create_stewardship(
    req: Request<Incoming>,
    pool: &DbPool,
    node_id: &str,
) -> Result<Response<Full<Bytes>>, StorageError> {
    let body = read_body(req).await?;
    let mut input: CreateNodeStewardshipInputView = serde_json::from_slice(&body)
        .map_err(|e| StorageError::InvalidInput(format!("Invalid stewardship input: {}", e)))?;

    // Override node_id from URL path
    input.node_id = node_id.to_string();

    let db_input: stewarded_nodes::CreateNodeStewardshipInput = input.into();
    let mut conn = pool.get().map_err(|e| StorageError::Internal(e.to_string()))?;
    let stewardship = stewarded_nodes::create_node_stewardship(&mut conn, db_input)?;

    let view: NodeStewardshipView = stewardship.into();
    response::created(&view)
}
```

### Task 8: Seeder Script — `seed-nodes.ts`

**Files:**
- Create: `genesis/seeder/src/seed-nodes.ts`
- Modify: `genesis/seeder/package.json` (add scripts)

**Step 1: Create `seed-nodes.ts`**

Create `genesis/seeder/src/seed-nodes.ts`. This script:
1. Reads `genesis/data/shefa/nodes.json`
2. For each node with a `context` field, creates a ContentNode via `/db/content/bulk` to get an EPR ID
3. Registers nodes via `POST /db/nodes`
4. Creates stewardship relationships via `POST /db/nodes/{id}/stewardship`

```typescript
#!/usr/bin/env tsx
/**
 * Seed Stewarded Nodes — register nodes and stewardship relationships.
 *
 * Reads from genesis/data/shefa/nodes.json.
 * Creates EPR-addressed context ContentNodes for natural language descriptions,
 * then registers nodes and stewardship relationships via elohim-storage API.
 *
 * Prerequisites: seed-humans must have run first (stewardship references human IDs).
 */

import { readFileSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

const __dirname = dirname(fileURLToPath(import.meta.url));

interface NodeSeed {
  id: string;
  displayName: string;
  cpuCores: number;
  memoryGb: number;
  storageTb: number;
  bandwidthMbps: number;
  stewardTier: string;
  region: string;
  context?: string;
}

interface StewardshipSeed {
  nodeId: string;
  humanId: string;
  affinityScore: number;
  relationship: string;
  context?: string;
}

interface NodesJson {
  nodes: NodeSeed[];
  stewardship: StewardshipSeed[];
}

async function main(): Promise<void> {
  const storageUrl = (
    process.env.STORAGE_URL || 'http://localhost:8090'
  ).replace(/\/$/, '');

  const jsonPath = resolve(__dirname, '../../data/shefa/nodes.json');
  const data: NodesJson = JSON.parse(readFileSync(jsonPath, 'utf-8'));

  console.log(`\nSeeding ${data.nodes.length} stewarded nodes...\n`);

  // Step 1: Create context ContentNodes for nodes and stewardship entries
  const contextItems: Array<{
    id: string;
    title: string;
    contentType: string;
    contentFormat: string;
    contentBody: string;
    reach: string;
  }> = [];

  for (const node of data.nodes) {
    if (node.context) {
      contextItems.push({
        id: `context-node-${node.id}`,
        title: `Node Context: ${node.displayName}`,
        contentType: 'node-context',
        contentFormat: 'text',
        contentBody: node.context,
        reach: 'intimate',
      });
    }
  }

  for (const s of data.stewardship) {
    if (s.context) {
      contextItems.push({
        id: `context-stewardship-${s.nodeId}-${s.humanId}`,
        title: `Stewardship Context: ${s.nodeId} ↔ ${s.humanId}`,
        contentType: 'stewardship-context',
        contentFormat: 'text',
        contentBody: s.context,
        reach: 'intimate',
      });
    }
  }

  if (contextItems.length > 0) {
    const contextRes = await fetch(`${storageUrl}/db/content/bulk`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'X-Schema-Version': '1' },
      body: JSON.stringify(contextItems),
    });
    if (!contextRes.ok) {
      const err = await contextRes.text();
      console.error(`  [X] Failed to seed context content: ${err}`);
    } else {
      const result = await contextRes.json();
      console.log(
        `  [+] Context content: ${result.inserted} inserted, ${result.skipped} skipped`
      );
    }
  }

  // Step 2: Register nodes
  for (const node of data.nodes) {
    const contextEprId = node.context ? `context-node-${node.id}` : undefined;

    const res = await fetch(`${storageUrl}/db/nodes`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        id: node.id,
        displayName: node.displayName,
        claimStatus: 'claimed',
        cpuCores: node.cpuCores,
        memoryGb: node.memoryGb,
        storageTb: node.storageTb,
        bandwidthMbps: node.bandwidthMbps,
        stewardTier: node.stewardTier,
        custodianOptIn: true,
        region: node.region,
        contextEprId: contextEprId,
      }),
    });

    if (res.ok) {
      console.log(`  [+] ${node.displayName.padEnd(20)} ${node.id}`);
    } else if (res.status === 409) {
      console.log(`  [=] ${node.displayName.padEnd(20)} ${node.id} (exists)`);
    } else {
      const err = await res.text();
      console.log(`  [X] ${node.displayName.padEnd(20)} ${err}`);
    }
  }

  // Step 3: Create stewardship relationships
  console.log(`\nSeeding ${data.stewardship.length} stewardship relationships...\n`);

  for (const s of data.stewardship) {
    const contextEprId = s.context
      ? `context-stewardship-${s.nodeId}-${s.humanId}`
      : undefined;

    const res = await fetch(`${storageUrl}/db/nodes/${s.nodeId}/stewardship`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        nodeId: s.nodeId,
        humanId: s.humanId,
        affinityScore: s.affinityScore,
        relationship: s.relationship,
        contextEprId: contextEprId,
      }),
    });

    const icon = res.ok ? '+' : res.status === 409 ? '=' : 'X';
    const label = `${s.humanId} → ${s.nodeId} (${s.relationship})`;
    if (!res.ok && res.status !== 409) {
      const err = await res.text();
      console.log(`  [${icon}] ${label}: ${err}`);
    } else {
      console.log(`  [${icon}] ${label}`);
    }
  }

  console.log('\nDone.\n');
}

main().catch((err) => {
  console.error('Fatal:', err);
  process.exit(1);
});
```

```typescript
/**
 * Aggregated availability across all stewarded nodes.
 * Three numbers per resource: total capacity, committed, available to share.
 */
export interface NodeAvailability {
  cpu: ResourceAvailability;
  memory: ResourceAvailability;
  storage: ResourceAvailability;
  bandwidth: ResourceAvailability;
}

export interface ResourceAvailability {
  total: number;
  committed: number;  // Constitutional limits — zero for now
  available: number;  // Total minus committed
  unit: string;       // 'cores', 'GB', 'TB', 'Mbps'
}

/**
 * Calculate aggregate availability from stewarded nodes.
 */
export function calculateNodeAvailability(nodes: StewardedNode[]): NodeAvailability {
  const claimedNodes = nodes.filter(n => n.claimStatus === 'claimed');

  const totalCpu = claimedNodes.reduce((sum, n) => sum + n.cpuCores, 0);
  const totalMemory = claimedNodes.reduce((sum, n) => sum + n.memoryGb, 0);
  const totalStorage = claimedNodes.reduce((sum, n) => sum + n.storageTb, 0);
  const totalBandwidth = claimedNodes.reduce((sum, n) => sum + n.bandwidthMbps, 0);

  return {
    cpu: { total: totalCpu, committed: 0, available: totalCpu, unit: 'cores' },
    memory: { total: totalMemory, committed: 0, available: totalMemory, unit: 'GB' },
    storage: { total: totalStorage, committed: 0, available: totalStorage, unit: 'TB' },
    bandwidth: { total: totalBandwidth, committed: 0, available: totalBandwidth, unit: 'Mbps' },
  };
}
```

```typescript
import { inject, Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';

import { StorageClientService } from '@app/elohim';
import {
  StewardedNode,
  NodeStewardshipSummary,
  NodeAvailability,
  calculateNodeAvailability,
} from '../models/shefa-dashboard.model';

/**
 * Thin HTTP client for stewarded node topology.
 * Reads from elohim-storage projection — fast and queryable.
 */
@Injectable({ providedIn: 'root' })
export class NodeTopologyApiService {
  private readonly http = inject(HttpClient);
  private readonly storage = inject(StorageClientService);

  /** List all stewarded nodes (optionally filtered by claim status). */
  listNodes(claimStatus?: string): Observable<StewardedNode[]> {
    const params = claimStatus ? { claimStatus } : {};
    return this.http.get<StewardedNode[]>(
      `${this.storage.baseUrl}/db/nodes`,
      { params }
    );
  }

  /** Get a single node with its stewardship relationships. */
  getNode(nodeId: string): Observable<StewardedNode> {
    return this.http.get<StewardedNode>(
      `${this.storage.baseUrl}/db/nodes/${nodeId}`
    );
  }

  /** Calculate aggregate availability from all claimed nodes. */
  getAvailability(nodes: StewardedNode[]): NodeAvailability {
    return calculateNodeAvailability(nodes);
  }
}
```
